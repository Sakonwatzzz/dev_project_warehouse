// public/page/sale/script/sale_bills.js

// ใช้ตัวแปรจาก window.saleBillsData
let bills = window.saleBillsData.bills || [];
let currentSaleId = window.saleBillsData.currentSaleId || null;

function switchBill(saleId) {
    if (saleId === currentSaleId) return;
    $.ajax({
        url: 'page/sale/ajax/ajax_switch_sale.php',
        type: 'POST',
        data: {
            action: 'switch_sale',
            sale_id: saleId
        },
        dataType: 'json',
        success: function (response) {
            if (response.status === 'success') {
                currentSaleId = saleId;
                updateTabStyles();
                window.location.href = `?page=sale_add&sale_id=${saleId}`;
            } else {
                Swal.fire({
                    icon: 'error',
                    title: 'ข้อผิดพลาด',
                    text: response.message,
                    confirmButtonText: 'ตกลง'
                });
            }
        },
        error: function (xhr) {
            Swal.fire({
                icon: 'error',
                title: 'ข้อผิดพลาด',
                text: xhr.responseText || 'เกิดข้อผิดพลาดในการสลับบิล',
                confirmButtonText: 'ตกลง'
            });
        }
    });
}

function cancelSale(saleId) {
    event.stopPropagation();
    Swal.fire({
        title: 'ยืนยันการยกเลิกบิล?',
        icon: 'warning',
        showCancelButton: true,
        confirmButtonText: 'ยกเลิกบิล',
        cancelButtonText: 'ยกเลิก'
    }).then((result) => {
        if (result.isConfirmed) {
            $.ajax({
                url: 'page/sale/ajax/ajax_cancel_sale.php',
                type: 'POST',
                data: {
                    action: 'cancel_sale',
                    sale_id: saleId
                },
                dataType: 'json',
                success: function (response) {
                    if (response.status === 'success') {
                        Swal.fire({
                            icon: 'success',
                            title: 'สำเร็จ',
                            text: response.message,
                            timer: 1500
                        }).then(() => {
                            window.location.href = response.redirect;
                        });
                    } else {
                        Swal.fire({
                            icon: 'error',
                            title: 'ข้อผิดพลาด',
                            text: response.message
                        });
                    }
                },
                error: function (xhr) {
                    Swal.fire({
                        icon: 'error',
                        title: 'ข้อผิดพลาด',
                        text: xhr.responseText || 'ไม่สามารถยกเลิกบิลได้'
                    });
                }
            });
        }
    });
}

function updateTabStyles() {
    $('#billTabs > div').each(function () {
        const tabId = $(this).attr('id').replace('tab_', '');
        if (tabId == currentSaleId) {
            $(this).removeClass('tab-inactive').addClass('tab-active');
        } else {
            $(this).removeClass('tab-active').addClass('tab-inactive');
        }
    });
}

$('#new_sale').on('click', function () {
    $.ajax({
        url: 'page/sale/ajax/ajax_new_sale.php',
        type: 'POST',
        data: {
            action: 'new_sale'
        },
        dataType: 'json',
        success: function (response) {
            if (response.status === 'success') {
                bills.push(response.sale_id);
                currentSaleId = response.sale_id;
                Toast.fire({
                    icon: 'success',
                    title: response.message
                });
                window.location.href = response.redirect;
            } else {
                Swal.fire({
                    icon: 'error',
                    title: 'ข้อผิดพลาด',
                    text: response.message
                });
            }
        },
        error: function (xhr) {
            Swal.fire({
                icon: 'error',
                title: 'ข้อผิดพลาด',
                text: xhr.responseText || 'ไม่สามารถสร้างบิลใหม่ได้'
            });
        }
    });
});
