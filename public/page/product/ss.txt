    $query = "SELECT
        product.product_id, 
        product.product_barcode, 
        product.product_name, 
        product.product_price1, 
        product.product_price2, 
        product.product_price3, 
        product.product_price4, 
        product.product_capital, 
        product.product_type_id, 
        product.product_brand_id,
        product_brand.product_brand_name, 
        product_type.product_type_name
    FROM
        product
    INNER JOIN
        product_brand ON product.product_brand_id = product_brand.product_brand_id
    INNER JOIN
        product_type ON product.product_type_id = product_type.product_type_id";

        $searchTerm = $_GET['search'];
        $query = $query . " WHERE product.product_name LIKE :searchTerm ORDER BY product.product_id ASC";
        $stmt = $pdo->prepare($query);
        $stmt->execute(['searchTerm' => '%' . $searchTerm . '%']);
        $productResults = $stmt;
    } else {
        $query = $query . " ORDER BY product.product_id ASC";
        $productResults = $pdo->query($query);
    }

    // Handle delete
    if (isset($_GET['id']) && !empty($_GET['id'])) {
        $id = filter_var($_GET['id'], FILTER_VALIDATE_INT);
        if ($id !== false) {
            $stmt = $pdo->prepare("DELETE FROM product WHERE product_id = ?");
            $stmt->execute([$id]);
            header('Location: /index.php?page=product_list');
            exit;
        }
    }

    // Query for product types
    $queryTypes = "SELECT product_type_id, product_type_name FROM product_type";
    $typeResults = $pdo->query($queryTypes);
    if (!$typeResults) {
        throw new PDOException("Failed to fetch product types");
    }

    // Query for product brands
    $queryBrands = "SELECT product_brand_id, product_brand_name FROM product_brand";
    $brandResults = $pdo->query($queryBrands);
    if (!$brandResults) {
        throw new PDOException("Failed to fetch product brands");
    }

    <script>
        function searchTable() {
            const input = document.getElementById("searchInput");
            const filter = input.value.toLowerCase();
            const table = document.getElementById("dataTable");
            const tr = table.getElementsByTagName("tr");
            let foundAny = false;

            for (let i = 1; i < tr.length; i++) {
                let found = false;
                const td = tr[i].getElementsByTagName("td");

                for (let j = 0; j < td.length; j++) {
                    if (td[j]) {
                        const txtValue = td[j].textContent || td[j].innerText;
                        if (txtValue.toLowerCase().indexOf(filter) > -1) {
                            found = true;
                            break;
                        }
                    }
                }

                tr[i].style.display = found ? "" : "none";
                if (found) foundAny = true;
            }

            const noDataMessage = document.getElementById("noDataMessage");
            if (!foundAny) {
                if (!noDataMessage) {
                    const message = document.createElement("p");
                    message.id = "noDataMessage";
                    message.textContent = "ไม่พบข้อมูล";
                    message.style.cssText = "color: red; font-weight: bold; margin-top: 10px; text-align: center;";
                    table.parentNode.insertBefore(message, table.nextSibling);
                }
            } else if (noDataMessage) {
                noDataMessage.remove();
            }
        }

        function confirmDelete(id) {
            if (confirm('คุณต้องการลบสินค้านี้ใช่หรือไม่?')) {
                window.location.href = `page/product/product_list.php?id=${id}`;
            }
        }

        document.addEventListener("DOMContentLoaded", function() {
            const urlParams = new URLSearchParams(window.location.search);
            const searchInput = document.getElementById("searchInput");
            searchInput.focus();

            if (urlParams.has('success')) {
                const notification = document.getElementById('notification');
                notification.style.display = 'block';

                setTimeout(() => {
                    notification.style.display = 'none';
                }, 2000);
            }
        });
    </script>
